cmake_minimum_required(VERSION 3.16)
project(sondechase VERSION 0.4.1 LANGUAGES C)
# The store reads the version from app-builder.json and the package from this project; keep them equal.
if(NOT CMAKE_VERSION VERSION_LESS 3.19)
  file(READ "${CMAKE_SOURCE_DIR}/app-builder.json" SC_APP_BUILDER)
  string(JSON SC_STORE_VERSION GET "${SC_APP_BUILDER}" version)
  if(NOT SC_STORE_VERSION STREQUAL PROJECT_VERSION)
    message(FATAL_ERROR "app-builder.json version ${SC_STORE_VERSION} != project version ${PROJECT_VERSION}")
  endif()
endif()
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(SC_BUILD_UI "Build the LVGL application (fetches LVGL v9.5.0)" ON)
option(SC_SIM "Desktop build: SDL2 window instead of /dev/fb0 + evdev" OFF)

# Core: no LVGL dependency, unit-tested on any Linux host.
add_library(sc_core STATIC src/base.c src/geo.c src/io.c src/track.c src/app.c src/demo.c src/tiles.c src/sdr.c src/dsp.c src/radio.c)
target_include_directories(sc_core PUBLIC src)
target_compile_definitions(sc_core PUBLIC SC_VERSION="${PROJECT_VERSION}"
  SC_DECODER_DIR="/usr/lib/${PROJECT_NAME}") # where the package installs fsk_demod and rs41mod
target_compile_options(sc_core PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
target_link_libraries(sc_core PUBLIC m Threads::Threads ${CMAKE_DL_LIBS}) # sdr.c: USB thread, dlopen(librtlsdr)

enable_testing()
add_executable(test_core tests/test_core.c src/keys.c) # keys.c: evdev only, no LVGL
target_link_libraries(test_core PRIVATE sc_core)
# librtlsdr stand-in loaded by the tests through cfg rtlsdr_lib (never installed)
add_library(fake_rtlsdr SHARED tests/fake_rtlsdr.c)
target_link_libraries(fake_rtlsdr PRIVATE m)
add_test(NAME core COMMAND test_core ${CMAKE_SOURCE_DIR}/tests/fake_decoder.sh $<TARGET_FILE:fake_rtlsdr>)

option(SC_RENDER "Build tools/render: headless screenshots of every screen (demo data)" OFF)

if(SC_BUILD_UI)
  include(FetchContent)
  if(SC_RENDER)
    add_compile_definitions(SC_HEADLESS=1)
  elseif(SC_SIM)
    add_compile_definitions(SC_SIM=1)
  endif()
  set(LV_BUILD_CONF_PATH ${CMAKE_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
  set(LV_CONF_PATH ${CMAKE_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
  FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.5.0 GIT_SHALLOW TRUE)
  # Same pattern as the CardputerZero Template: EXCLUDE_FROM_ALL keeps LVGL's own install rules
  # (headers, static library) out of the application package.
  FetchContent_GetProperties(lvgl)
  if(NOT lvgl_POPULATED)
    if(POLICY CMP0169)
      cmake_policy(PUSH)
      cmake_policy(SET CMP0169 OLD)
    endif()
    FetchContent_Populate(lvgl)
    if(POLICY CMP0169)
      cmake_policy(POP)
    endif()
  endif()
  add_subdirectory("${lvgl_SOURCE_DIR}" "${lvgl_BINARY_DIR}" EXCLUDE_FROM_ALL)

  if(SC_RENDER)
    add_executable(sc_render tools/render.c src/ui.c src/keys.c)
    target_compile_options(sc_render PRIVATE -Wall -Wextra)
    target_link_libraries(sc_render PRIVATE sc_core lvgl)
    return()
  endif()

  add_executable(sondechase src/main.c src/ui.c src/keys.c)
  target_compile_options(sondechase PRIVATE -Wall -Wextra)
  target_link_libraries(sondechase PRIVATE sc_core lvgl)
  if(SC_SIM)
    find_package(SDL2 REQUIRED)
    target_link_libraries(lvgl PUBLIC SDL2::SDL2)
  endif()
  if(NOT SC_SIM)
    include(cmake/radiosonde-decoders.cmake) # fsk_demod + rs41mod, shipped in the package
  endif()

  include(cmake/cm0-package.cmake) # install layout and .deb packaging (CardputerZero Template contract)
endif()
